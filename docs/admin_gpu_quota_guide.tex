%==============================================================================
\section{Step 1: Request GPU Quota (5 minutes)}
%==============================================================================

\subsection{1.1 Open the Quotas Page}

Click this link (make sure you're logged in as project owner):

\begin{center}
\fbox{\url{https://console.cloud.google.com/iam-admin/quotas?project=piqc-483417}}
\end{center}

\subsection{1.2 Filter for GPU Quota}

\begin{enumerate}
    \item In the \textbf{Filter} box at the top, type: \texttt{GPUS\_ALL\_REGIONS}
    \item Press Enter
    \item You should see a row showing \textbf{"GPUs (all regions)"} with \textbf{Limit: 0}
\end{enumerate}

\subsection{1.3 Request Quota Increase}

\begin{enumerate}
    \item \textbf{Check the checkbox} next to "GPUs (all regions)"
    \item Click the \textbf{"EDIT QUOTAS"} button at the top of the page
    \item A side panel will open on the right
\end{enumerate}

\subsection{1.4 Fill in the Request Form}

\begin{table}[h]
\centering
\begin{tabular}{@{}lp{10cm}@{}}
\toprule
\textbf{Field} & \textbf{Value} \\
\midrule

Request description & Testing PIQC (AI/ML monitoring tool) on GKE with T4 GPU. Short-term testing only, will delete resources after testing. \\
\bottomrule
\end{tabular}
\end{table}

\subsection{1.5 Submit the Request}

\begin{enumerate}
    \item Click \textbf{"NEXT"}
    \item Verify your contact information
    \item Click \textbf{"SUBMIT REQUEST"}
\end{enumerate}

\subsection{1.6 Confirmation}

\begin{itemize}
    \item You'll see a confirmation message
    \item Google will email you when the quota is approved
    \item \textbf{Typical wait time: 24-48 hours} (sometimes faster)
